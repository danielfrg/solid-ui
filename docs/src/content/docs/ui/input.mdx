# Input

## Usage

Add the theme to your CSS:

```css
@import "tailwindcss";
@import "@danielfrg/solid-ui/theme.css";
```

Import and use the Input components:

```tsx
import { Input, InputField, InputLabel } from "@danielfrg/solid-ui/input"

<Input>
  <InputLabel>Email</InputLabel>
  <InputField type="email" placeholder="you@example.com" />
</Input>
```
