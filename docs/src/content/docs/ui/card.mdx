# Card

## Usage

Add the theme to your CSS:

```css
@import "tailwindcss";
@import "@danielfrg/solid-ui/theme.css";
```

Import and use the `Card` components:

```tsx
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from "@danielfrg/solid-ui/card"

<Card>
  <CardHeader>
    <CardTitle>Card Title</CardTitle>
    <CardDescription>Card description.</CardDescription>
  </CardHeader>
  <CardContent>
    <p>Content goes here.</p>
  </CardContent>
  <CardFooter>
    <p>Footer</p>
  </CardFooter>
</Card>
```
