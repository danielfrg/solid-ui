# Tooltip

## Usage

Add the theme to your CSS:

```css
@import "tailwindcss";
@import "@danielfrg/solid-ui/theme.css";
```

Import and use the `Tooltip` components:

```tsx
import { Tooltip, TooltipTrigger, TooltipContent } from "@danielfrg/solid-ui/tooltip"
import { Button } from "@danielfrg/solid-ui/button"
<Tooltip>
  <TooltipTrigger as={Button} variant="outline">
    Hover me
  </TooltipTrigger>
  <TooltipContent>
    <p>Tooltip content</p>
  </TooltipContent>
</Tooltip>
```
