# Accordion

## Usage

Add the theme to your CSS:

```css
@import "tailwindcss";
@import "@danielfrg/solid-ui-nova/theme.css";
```

Import and use the Accordion components:

```tsx
import { Accordion, AccordionItem, AccordionTrigger, AccordionContent } from "@danielfrg/solid-ui-nova/accordion"

;<Accordion defaultValue={["item-1"]}>
  <AccordionItem value="item-1">
    <AccordionTrigger>Is it accessible?</AccordionTrigger>
    <AccordionContent>Yes. It adheres to the WAI-ARIA design pattern.</AccordionContent>
  </AccordionItem>
</Accordion>
```
